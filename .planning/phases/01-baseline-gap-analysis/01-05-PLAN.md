```python
#!/usr/bin/env python3
"""Consolidate gap reports into baseline document"""
import json
from datetime import datetime
from pathlib import Path

reports_dir = Path(".planning/phases/01-baseline-gap-analysis/reports")

# Load individual reports
targets = {}
for name in ["cjson", "libpcap", "libaom"]:
    report_path = reports_dir / f"{name}_gap_report.json"
    if report_path.exists():
        with open(report_path) as f:
            targets[name] = json.load(f)

# Consolidate
consolidated = {
    "generated_at": datetime.now().isoformat(),
    "phase": "01-baseline-gap-analysis",
    "targets": {},
    "summary": {
        "total_apis_analyzed": 0,
        "total_with_constraints": 0,
        "gap_by_category": {}
    }
}

for name, report in targets.items():
    consolidated["targets"][name] = {
        "liberator_apis": report["counts"]["liberator_apis"],
        "with_constraints": report["counts"]["with_constraints"],
        "constraint_coverage_pct": report["coverage_pct"]["constraint_coverage"],
        "baseline_function_coverage": {
            "cjson": 76.99,
            "libpcap": 13.12,
            "libaom": 0.69
        }.get(name, 0),
        "tier2_gap_count": len(report["gaps"]["tier2_no_constraints"])
    }
    consolidated["summary"]["total_apis_analyzed"] += report["counts"]["liberator_apis"]
    consolidated["summary"]["total_with_constraints"] += report["counts"]["with_constraints"]

    # Aggregate categories
    if "categorized_gaps" in report:
        for cat, funcs in report["categorized_gaps"].get("tier2_by_category", {}).items():
            if cat not in consolidated["summary"]["gap_by_category"]:
                consolidated["summary"]["gap_by_category"][cat] = 0
            consolidated["summary"]["gap_by_category"][cat] += len(funcs)

# Write consolidated
with open(reports_dir / "consolidated_baseline.json", "w") as f:
    json.dump(consolidated, f, indent=2)
```
